zmc_setup_once = {
	set_global_variable = zmc_setup_done

	set_global_variable = {
		name = zmc_entire_world_real_size
		value = 0
	}

	#geography never changes so this only has to be done once
	every_ownable_location = { #for urbanisation suitability mapmode to work from the start
		set_variable = {
			name = zmc_local_urbanisation_suitability
			value = zmc_local_urbanisation_suitability_level
		}
		change_global_variable = {
			name = zmc_entire_world_real_size
			add = zmc_real_world_size
		}
	}

	#geography never changes so this only has to be done once
	every_province_definition = { #for urbanisation suitability mapmode to work from the start
		set_local_variable = {
			name = zmc_counting_helper
			value = 0
		}
		ordered_location_in_province_definition = {
			limit = { is_ownable = yes }
			order_by = var:zmc_local_urbanisation_suitability #zmc_local_urbanisation_suitability_level
			min = 0
			set_variable = {
				name = zmc_local_urbanisation_suitability_sorted
				value = 1
			}
			change_variable = { 
				name = zmc_local_urbanisation_suitability_sorted 
				add = local_var:zmc_counting_helper
			}
			province_definition = {
				change_local_variable = { 
					name = zmc_counting_helper 
					add = 1
				}
			}
		}
	}

	zmc_setup_yearly = yes
}



zmc_setup_yearly = {
	set_global_variable = {
		name = zmc_world_population_var
		value = 0
	}
	set_global_variable = {
		name = zmc_world_tax_base_var
		value = 0
	}
	every_ownable_location = {
		change_global_variable = {
			name = zmc_world_population_var
			add = {
				value = population
				multiply = 1000
			}
		}
		change_global_variable = {
			name = zmc_world_tax_base_var
			add = location_tax_base
		}
	}
	set_global_variable = { #calculating world tax base per population every year to save on performance
		name = zmc_world_tax_base_per_population_var
		value = {
			value = global_var:zmc_world_tax_base_var
			divide = global_var:zmc_world_population_var
			multiply = 1000
		}
	}

	zmc_calculate_higher_tier_stats = yes
}




#the following codes base was provided by caeservincens, but actual implementation for the mod was entirely made by github copilot, so have caution
zmc_calculate_higher_tier_stats = { #caches area/region/sub_continent/continent population & tax_base once per year instead of recalculating on every map paint
	clear_global_variable_map = zmc_area_population_map
	clear_global_variable_map = zmc_area_tax_base_map
	clear_global_variable_map = zmc_region_population_map
	clear_global_variable_map = zmc_region_tax_base_map
	clear_global_variable_map = zmc_sub_continent_population_map
	clear_global_variable_map = zmc_sub_continent_tax_base_map
	clear_global_variable_map = zmc_continent_population_map
	clear_global_variable_map = zmc_continent_tax_base_map

	every_continent = {
		set_local_variable = { name = zmc_continent_population_accumulator value = 0 }
		set_local_variable = { name = zmc_continent_tax_base_accumulator value = 0 }
		every_sub_continent_in_continent = {
			set_local_variable = { name = zmc_sub_continent_population_accumulator value = 0 }
			set_local_variable = { name = zmc_sub_continent_tax_base_accumulator value = 0 }
			every_region_in_continent = { #despite the name, this is scoped from sub_continent and iterates its regions
				set_local_variable = { name = zmc_region_population_accumulator value = 0 }
				set_local_variable = { name = zmc_region_tax_base_accumulator value = 0 }
				every_area_in_region = {
					set_local_variable = { name = zmc_area_population_accumulator value = 0 }
					set_local_variable = { name = zmc_area_tax_base_accumulator value = 0 }
					every_province_in_area = {
						every_location_in_province = {
							limit = { is_ownable = yes }
							change_local_variable = {
								name = zmc_area_population_accumulator
								add = population
							}
							change_local_variable = {
								name = zmc_area_tax_base_accumulator
								add = location_tax_base
							}
							change_local_variable = {
								name = zmc_region_population_accumulator
								add = population
							}
							change_local_variable = {
								name = zmc_region_tax_base_accumulator
								add = location_tax_base
							}
							change_local_variable = {
								name = zmc_sub_continent_population_accumulator
								add = population
							}
							change_local_variable = {
								name = zmc_sub_continent_tax_base_accumulator
								add = location_tax_base
							}
							change_local_variable = {
								name = zmc_continent_population_accumulator
								add = population
							}
							change_local_variable = {
								name = zmc_continent_tax_base_accumulator
								add = location_tax_base
							}
						}
					}
					add_to_global_variable_map = {
						name = zmc_area_population_map
						key = this
						value = local_var:zmc_area_population_accumulator
					}
					add_to_global_variable_map = {
						name = zmc_area_tax_base_map
						key = this
						value = local_var:zmc_area_tax_base_accumulator
					}
				}
				add_to_global_variable_map = {
					name = zmc_region_population_map
					key = this
					value = local_var:zmc_region_population_accumulator
				}
				add_to_global_variable_map = {
					name = zmc_region_tax_base_map
					key = this
					value = local_var:zmc_region_tax_base_accumulator
				}
			}
			add_to_global_variable_map = {
				name = zmc_sub_continent_population_map
				key = this
				value = local_var:zmc_sub_continent_population_accumulator
			}
			add_to_global_variable_map = {
				name = zmc_sub_continent_tax_base_map
				key = this
				value = local_var:zmc_sub_continent_tax_base_accumulator
			}
		}
		add_to_global_variable_map = {
			name = zmc_continent_population_map
			key = this
			value = local_var:zmc_continent_population_accumulator
		}
		add_to_global_variable_map = {
			name = zmc_continent_tax_base_map
			key = this
			value = local_var:zmc_continent_tax_base_accumulator
		}
	}
}
