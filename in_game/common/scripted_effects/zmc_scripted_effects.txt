zmc_setup_once = {
	set_global_variable = zmc_setup_done

	set_global_variable = {
		name = zmc_entire_world_real_size
		value = 0
	}

	#geography never changes so this only has to be done once
	every_location_in_the_world = { #for urbanisation suitability mapmode to work from the start
		limit = { is_ownable = yes }
		set_variable = {
			name = zmc_local_urbanisation_suitability
			value = zmc_local_urbanisation_suitability_level
		}
		change_global_variable = {
			name = zmc_entire_world_real_size
			add = zmc_real_world_size
		}
	}

	#geography never changes so this only has to be done once
	every_province_definition = { #for urbanisation suitability mapmode to work from the start
		set_local_variable = {
			name = zmc_counting_helper
			value = 0
		}
		ordered_location_in_province_definition = {
			limit = { is_ownable = yes }
			order_by = var:zmc_local_urbanisation_suitability #zmc_local_urbanisation_suitability_level
			min = 0
			set_variable = {
				name = zmc_local_urbanisation_suitability_sorted
				value = 1
			}
			change_variable = { 
				name = zmc_local_urbanisation_suitability_sorted 
				add = local_var:zmc_counting_helper
			}
			province_definition = {
				change_local_variable = { 
					name = zmc_counting_helper 
					add = 1
				}
			}
		}
	}

	zmc_setup_yearly = yes
}



zmc_setup_yearly = {
	set_global_variable = { #calculating world population every year to save on performance
		name = zmc_world_population
		value = {
			every_location_in_the_world = {
				limit = { is_ownable = yes }
				add = population
			}
			multiply = 1000
		}
	}
	set_global_variable = { #calculating world tax base every year to save on performance
		name = zmc_world_tax_base
		value = {
			every_location_in_the_world = {
				limit = { is_ownable = yes }
				add = location_tax_base
			}
		}
	}
	set_global_variable = { #calculating world tax base per population every year to save on performance
		name = zmc_world_tax_base_per_population_var
		value = {
			value = global_var:zmc_world_tax_base
			divide = global_var:zmc_world_population
			multiply = 1000
		}
	}
}
