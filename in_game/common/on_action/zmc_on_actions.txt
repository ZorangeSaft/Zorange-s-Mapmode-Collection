#####zmc

on_game_start = { #for new saves
	on_actions = {
		zmc_on_game_start
	}
}

monthly_country_pulse = { #for adding to existing saves
	on_actions = {
		zmc_on_game_start
	}
}



zmc_on_game_start = {
	trigger = {
		NOT = { has_global_variable = zmc_setup_done }
	}
	zmc_setup_once = yes #includes yearly
}



yearly_country_pulse = {
	on_actions = {
		zmc_yearly
	}
}



zmc_yearly = {
	effect = {
		zmc_setup_yearly = yes
	}
}
