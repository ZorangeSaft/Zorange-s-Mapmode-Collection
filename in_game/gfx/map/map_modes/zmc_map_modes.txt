zmc_building_cap_percentage = {
	map_color = {
		if = {
			limit = {
				is_land = no
			}
			value = hsv { 0.58 0.50 0.52 } #figured out via trail and error from the color the basegame tells you, matches basegame color for water, altough water for some reason falls under fallback
		}
		else_if = {
			limit = {
				zmc_used_building_slots_percentage <= 5 #500%, so way over cap
				zmc_used_building_slots_percentage >= 1 #1 #100%
			}
			lerp = {
				max_color = rgb { 0 0 0 } #black
				min_color = rgb { 255 0 255 } #purple
				factor = {
					value = zmc_used_building_slots_percentage
					divide = 5
				}
			}
		}
		else_if = {
			limit = {
				zmc_used_building_slots_percentage < 1 #100%
				zmc_used_building_slots_percentage >= 0.5 #50%
			}
