		}
		else_if = {
			limit = {
				zmc_pop_growth_percentage >= 5 #5% and above
			}
			value = rgb { 255 0 255 } #purple
		}
		else_if = {
			limit = {
				zmc_pop_growth_percentage < 5
				zmc_pop_growth_percentage >= 3 #3% to 5%
			}
			lerp = {
				max_color = rgb { 255 0 255 } #purple
				min_color = rgb { 0 0 200 } #dark blue
				factor = {
					value = zmc_pop_growth_percentage
					subtract = 3
					divide = 2
				}
			}
		}
		else_if = {
			limit = {
				zmc_pop_growth_percentage < 3
				zmc_pop_growth_percentage >= 2 #2% to 3%
			}
			lerp = {
				max_color = rgb { 0 0 200 } #dark blue
				min_color = rgb { 135 206 250 } #light blue
				factor = {
					value = zmc_pop_growth_percentage
					subtract = 2
					divide = 1
				}
			}
		}
		else_if = {
			limit = {
				zmc_pop_growth_percentage < 2
				zmc_pop_growth_percentage >= 1 #1% to 2%
			}
			lerp = {
				max_color = rgb { 135 206 250 } #light blue
				min_color = rgb { 0 255 0 } #green
				factor = {
					value = zmc_pop_growth_percentage
					subtract = 1
					divide = 1
				}
			}
		}
		else_if = {
			limit = {
				zmc_pop_growth_percentage < 1
				zmc_pop_growth_percentage >= 0.5 #0.5% to 1%
			}
			lerp = {
				max_color = rgb { 0 255 0 } #green
				min_color = rgb { 255 255 0 } #yellow
				factor = {
					value = zmc_pop_growth_percentage
					subtract = 0.5
					divide = 0.5
				}
			}
		}
		else_if = {

				zmc_pop_growth_percentage < 0.5

			limit = {
				zmc_pop_growth_percentage < 0
				zmc_pop_growth_percentage >= -3 #-1% to 0%
			}
			lerp = { #colors have to be reversed for negative numbers
				max_color = rgb { 0 0 0 } #black
				min_color = rgb { 150 0 0 } #dark red
				factor = {
					value = zmc_pop_growth_percentage
					multiply = -1
					divide = 3
				}
			}
		}
		else_if = {
			limit = {
				NOT = {
					exists = owner
				}
			}
			value = rgb { 50 50 50 } #gray for uncolonised land
		}
		else = {
			value = rgb { 0 0 0 } #black for values below -3%
		}
	}

			limit = {
				is_land = no
			}
			value = hsv { 0.58 0.50 0.52 } #figured out via trail and error from the color the basegame tells you, matches basegame color for water, altough water for some reason falls under fallback
		}
		else_if = {
			limit = {
				zmc_local_monthly_development >= 5 #over +0.05 development growth per month
			}
			value = rgb { 255 0 255 } #purple
		}
		else_if = {
			limit = {
				zmc_local_monthly_development < 5 
				zmc_local_monthly_development >= 3 #over +0.03 development growth per month
			}
			lerp = {
				max_color = rgb { 255 0 255 } #purple
				min_color = rgb { 0 0 200 } #dark blue
				factor = {
					value = zmc_local_monthly_development
					subtract = 3
					divide = 2
				}
			}
		}
		else_if = {
			limit = {
				zmc_local_monthly_development < 3
				zmc_local_monthly_development >= 2 #over +0.02 development growth per month
			}
			lerp = {
				max_color = rgb { 0 0 200 } #dark blue
				min_color = rgb { 135 206 250 } #light blue
				factor = {
					value = zmc_local_monthly_development
					subtract = 2
				}
			}
		}
		else_if = {
			limit = {
				zmc_local_monthly_development < 2
				zmc_local_monthly_development >= 1 #over +0.01 development growth per month
			}
			lerp = {
				max_color = rgb { 135 206 250 } #light blue
				min_color = rgb { 0 255 0 } #green
				factor = {
					value = zmc_local_monthly_development
					subtract = 1
				}
			}
		}
		else_if = {
			limit = {
				zmc_local_monthly_development < 1
				zmc_local_monthly_development >= 0.5 #over +0.05 development growth per month
			}
			lerp = {
				max_color = rgb { 0 255 0 } #green
				min_color = rgb { 255 255 0 } #yellow
				factor = {
					value = zmc_local_monthly_development
					subtract = 0.5
					divide = 0.5
				}
			}
		}
		else_if = {
			limit = {
				zmc_local_monthly_development < 0.5
				zmc_local_monthly_development >= 0 #+0.00 development growth per month
				exists = owner #to make it so uncolonised land doesnt show up red
			}
			lerp = {
				max_color = rgb { 255 255 0 } #yellow
				min_color = rgb { 255 0 0 } #red
				factor = {
					value = zmc_local_monthly_development
					divide = 0.5
				}
			}
		}
		else_if = {
			limit = {
				zmc_local_monthly_development < 0
				zmc_local_monthly_development >= -3 #under -0.03 development growth per month
			}
			lerp = {
				max_color = rgb { 0 0 0 } #black
				min_color = rgb { 150 0 0 } #dark red
				factor = {
					value = zmc_local_monthly_development
					multiply = -1
					divide = 3
				}
			}
		}
		else_if = {
			limit = {
				NOT = {
					exists = owner
				}
			}
			value = rgb { 50 50 50 } #gray for uncolonised land
		}
		else = {
			value = rgb { 0 0 0 } #black as fallback and out of bounds
		}
	}
